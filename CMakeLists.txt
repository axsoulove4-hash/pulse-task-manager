cmake_minimum_required(VERSION 3.20)
project(Pulse VERSION 1.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTORCC ON)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Pulse requires Linux procfs and sysfs")
endif()
find_package(Qt6 6.2 REQUIRED COMPONENTS Widgets Svg)
add_executable(pulse main.cpp resources.qrc)
target_link_libraries(pulse PRIVATE Qt6::Widgets Qt6::Svg)
target_compile_options(pulse PRIVATE -Wall -Wextra)
install(TARGETS pulse RUNTIME DESTINATION bin)
